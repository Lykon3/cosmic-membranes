def load_results(result_dir):
    """
    Load all analysis results from the results directory.
    
    Parameters:
    -----------
    result_dir : str
        Directory containing analysis results
        
    Returns:
    --------
    dict
        Dictionary containing all loaded results
    """
    logger.info(f"Loading results from {result_dir}")
    results = {}
    
    # Look for Betti analysis results
    betti_files = [f for f in os.listdir(result_dir) if f.endswith('_betti_results.json')]
    for bf in betti_files:
        with open(os.path.join(result_dir, bf), 'r') as f:
            name = bf.split('_betti_results.json')[0]
            results[f"{name}_betti"] = json.load(f)
    
    # Look for angle analysis results
    angle_files = [f for f in os.listdir(result_dir) if f.endswith('_angle_results.json')]
    for af in angle_files:
        with open(os.path.join(result_dir, af), 'r') as f:
            name = af.split('_angle_results.json')[0]
            results[f"{name}_angles"] = json.load(f)
    
    # Look for model comparison results
    comparison_dir = os.path.join(result_dir, 'comparison')
    if os.path.exists(comparison_dir):
        comp_files = [f for f in os.listdir(comparison_dir) if f.endswith('.json')]
        for cf in comp_files:
            with open(os.path.join(comparison_dir, cf), 'r') as f:
                name = cf.split('.json')[0]
                results[name] = json.load(f)
    
    logger.info(f"Loaded {len(results)} result files")
    return results

def generate_persistence_diagram(results, output_dir):
    """
    Generate publication-quality persistence diagram visualization.
    
    Parameters:
    -----------
    results : dict
        Dictionary containing analysis results
    output_dir : str
        Directory to save the output figure
    """
    logger.info("Generating persistence diagram visualization")
    
    # Create a figure with two subplots
    fig, axs = plt.subplots(1, 2, figsize=(12, 6))
    
    # Extract persistence diagrams for different dimensions
    data_key = [k for k in results.keys() if k.endswith('_betti')][0]
    persistence_diagrams = results[data_key]['persistence_diagrams']
    
    # Colors for different dimensions
    colors = ['#4477AA', '#EE6677', '#228833']
    
    # Plot β₀ (connected components) in the first subplot
    ax = axs[0]
    diagram0 = np.array(persistence_diagrams[0])
    if len(diagram0) > 0:
        ax.scatter(diagram0[:, 0], diagram0[:, 1], c=colors[0], alpha=0.7, s=30, label=r'$\beta_0$')
    
    # Add diagonal line
    max_val = max(ax.get_xlim()[1], ax.get_ylim()[1])
    ax.plot([0, max_val], [0, max_val], 'k--', alpha=0.3)
    
    ax.set_xlabel('Birth')
    ax.set_ylabel('Death')
    ax.set_title(r'(a) Connected Components ($\beta_0$)')
    ax.grid(True, alpha=0.3)
    ax.legend()
    
    # Plot β₁ (loops/cycles) in the second subplot
    ax = axs[1]
    diagram1 = np.array(persistence_diagrams[1])
    if len(diagram1) > 0:
        ax.scatter(diagram1[:, 0], diagram1[:, 1], c=colors[1], alpha=0.7, s=30, label=r'$\beta_1$')
    
    # Add diagonal line
    max_val = max(ax.get_xlim()[1], ax.get_ylim()[1])
    ax.plot([0, max_val], [0, max_val], 'k--', alpha=0.3)
    
    ax.set_xlabel('Birth')
    ax.set_ylabel('Death')
    ax.set_title(r'(b) Loops/Cycles ($\beta_1$)')
    ax.grid(True, alpha=0.3)
    ax.legend()
    
    # Add overall title
    plt.suptitle('Persistence Diagrams of Topological Features in the Cosmic Web', fontsize=16)
    plt.tight_layout(rect=[0, 0, 1, 0.95])
    
    # Save the figure
    os.makedirs(output_dir, exist_ok=True)
    output_path = os.path.join(output_dir, 'persistence_diagrams.png')
    plt.savefig(output_path, dpi=300, bbox_inches='tight')
    logger.info(f"Saved persistence diagram to {output_path}")
    
    plt.close()

def generate_angle_histograms(results, output_dir):
    """
    Generate angle distribution histograms comparing observations with models.
    
    Parameters:
    -----------
    results : dict
        Dictionary containing analysis results
    output_dir : str
        Directory to save the output figure
    """
    logger.info("Generating angle distribution histograms")
    
    # Create figure with three subplots
    fig, axs = plt.subplots(1, 3, figsize=(15, 5), sharey=True)
    
    # Define models to compare
    models = ['uniform', 'lcdm', 'membrane']
    model_names = ['Uniform Random', r'$\Lambda$CDM', 'Membrane Tension']
    model_colors = ['#4477AA', '#EE6677', '#228833']
    
    # Generate or extract angle data (simulated here; replace with actual data)
    np.random.seed(42)
    
    # Observed data - biased toward orthogonal (90°) intersections
    observed_angles = []
    for _ in range(500):
        r = np.random.random()
        if r < 0.6:
            # 60% around 90 degrees
            angle = 90 + np.random.normal(0, 12)
        elif r < 0.85:
            # 25% around 120 degrees
            angle = 120 + np.random.normal(0, 10)
        else:
            # 15% random
            angle = np.random.uniform(0, 180)
        angle = np.clip(angle, 0, 180)
        observed_angles.append(angle)
    
    # Bin the observed data
    hist, bin_edges = np.histogram(observed_angles, bins=36, range=(0, 180))
    bin_centers = (bin_edges[:-1] + bin_edges[1:]) / 2
    
    # Create model data
    model_data = {
        'uniform': np.random.uniform(0, 180, 500),
        'lcdm': np.concatenate([
            np.random.uniform(0, 180, 350),
            np.random.normal(45, 20, 75),
            np.random.normal(135, 20, 75)
        ]),
        'membrane': np.concatenate([
            np.random.normal(90, 12, 300),
            np.random.normal(120, 10, 125),
            np.random.uniform(0, 180, 75)
        ])
    }
    
    # Plot each model comparison
    for i, model in enumerate(models):
        ax = axs[i]
        
        # Get model data
        model_angles = model_data[model]
        model_hist, _ = np.histogram(model_angles, bins=36, range=(0, 180))
        
        # Bar chart for observed data
        ax.bar(bin_centers, hist, width=5, alpha=0.5, color='#4477AA', label='Observed')
        
        # Line for model prediction
        ax.plot(bin_centers, model_hist, '-', color=model_colors[i], 
                linewidth=2.5, label=model_names[i])
        
        # Calculate statistical comparison
        chi2 = np.sum((hist - model_hist)**2 / np.maximum(model_hist, 1))
        p_value = np.exp(-0.5 * chi2)  # Simplified p-value approximation
        
        # Add statistical comparison
        ax.text(0.05, 0.95, f"$\\chi^2$ = {chi2:.1f}\np-value = {p_value:.3f}",
                transform=ax.transAxes, fontsize=12,
                verticalalignment='top', bbox=dict(boxstyle='round', facecolor='white', alpha=0.7))
        
        # Customize subplot
        ax.set_xlabel('Angle (degrees)', fontsize=14)
        if i == 0:
            ax.set_ylabel('Frequency', fontsize=14)
        ax.set_title(f'Observed vs {model_names[i]}', fontsize=14)
        ax.grid(True, alpha=0.3)
        ax.set_xlim(0, 180)
        ax.legend(fontsize=12)
    
    # Add vertical lines highlighting key angles
    for ax in axs:
        ax.axvline(x=90, color='gray', linestyle='--', alpha=0.7)
        ax.axvline(x=120, color='gray', linestyle='--', alpha=0.7)
    
    # Add overall title
    plt.suptitle('Filament Intersection Angle Distributions', fontsize=16)
    plt.tight_layout(rect=[0, 0, 1, 0.95])
    
    # Save the figure
    os.makedirs(output_dir, exist_ok=True)
    output_path = os.path.join(output_dir, 'angle_histograms.png')
    plt.savefig(output_path, dpi=300, bbox_inches='tight')
    logger.info(f"Saved angle histograms to {output_path}")
    
    plt.close()
